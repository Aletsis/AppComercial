Private Sub FbtnConcepto_Click()

Dim Nombre As String
Dim Estatus As Long

    Call comFacturaMain.dbGetNoLock(hDbc, CONCEPTOS, IxCCODIGOC01, Range("CodigoConcepto").Value, lResultado)
    If lResultado = 0 Then
        'dbFieldNameChar hDbc, CONCEPTOS, "cNombreC01", Nombre, 255
        Call comFacturaMain.dbFieldNameChar(hDbc, CONCEPTOS, "cNombreC01", Nombre, 255, lResultado)
        Range("NombreConcepto").Value = Trim(fEliminaEspaciosNulos2(Nombre))
        'dbFieldNameLong hDbc, CONCEPTOS, "cEstatus", Estatus
        Call comFacturaMain.dbFieldNameLong(hDbc, CONCEPTOS, "cEstatus", Estatus, lResultado)
        If Estatus = 0 Then
            MsgBox "El concepto está inactivo", vbOKOnly
            Range("CodigoConcepto").Value = ""
            Range("FReferencia").Value = ""
        Else
            'dbFieldNameLong hDbc, CONCEPTOS, "cEsCFD", EsCFD
            'dbFieldNameLong hDbc, CONCEPTOS, "cVerFacEle", VersionCFD
            'dbFieldNameChar hDbc, CONCEPTOS, "cVerEsque", Esquema, 26
            Call comFacturaMain.dbFieldNameLong(hDbc, CONCEPTOS, "cEsCFD", EsCFD, lResultado)
            Call comFacturaMain.dbFieldNameLong(hDbc, CONCEPTOS, "cVerFacEle", VersionCFD, lResultado)
            Call comFacturaMain.dbFieldNameChar(hDbc, CONCEPTOS, "cVerEsque", Esquema, 26, lResultado)
            If EsCFD = 1 And VersionCFD = 2 And CDbl(Trim(fEliminaEspaciosNulos2(Esquema))) >= 3.3 Then
            Else
                MsgBox "El concepto debe de ser un CFDI versión 3.3", vbOKOnly
                Range("CodigoConcepto").Value = ""
                Range("FReferencia").Value = ""
            End If
            
            'Verificar si el concepto tiene habilitado el IEPS
            If gNumeroImpuestoIEPS <> 0 Then
                lImpuesto = "cUsaImpu0" + CStr(gNumeroImpuestoIEPS) + ""
                'dbFieldNameLong hDbc, CONCEPTOS, lImpuesto, lUsaImpuesto
                Call comFacturaMain.dbFieldNameLong(hDbc, CONCEPTOS, lImpuesto, lUsaImpuesto, lResultado)
                If (lUsaImpuesto = 1 And Worksheets("Parámetros").Range("HabilitarIEPS").Value = "Habilitar") Or (lUsaImpuesto <> 1 And Worksheets("Parámetros").Range("HabilitarIEPS").Value = "Sin uso") Then
                    MsgBox "El impuesto IEPS debe estar habilitado tanto en el concepto como en la hoja electrónica.", vbOKOnly
                    Range("CodigoConcepto").Value = ""
                    Range("FReferencia").Value = ""
                    Exit Sub
                End If
            End If
            
            If gNumeroSegundoImpuestoIEPS <> 0 Then
                lImpuesto = "cUsaImpu0" + CStr(gNumeroSegundoImpuestoIEPS) + ""
                'dbFieldNameLong hDbc, CONCEPTOS, lImpuesto, lUsaImpuesto
                Call comFacturaMain.dbFieldNameLong(hDbc, CONCEPTOS, lImpuesto, lUsaImpuesto, lResultado)
                If (lUsaImpuesto = 1 And Worksheets("Parámetros").Range("HabilitarIEPS").Value = "Habilitar") Or (lUsaImpuesto <> 1 And Worksheets("Parámetros").Range("HabilitarIEPS").Value = "Sin uso") Then

        Cells(1, kColImporteLocal).ColumnWidth = Cells(1, kColImpuestoLocal).ColumnWidth
        Cells(1, kColImporteIEPS).ColumnWidth = Cells(1, kColTasaCuotaIEPS1).ColumnWidth
        Cells(1, kColImporteSegundoIEPS).ColumnWidth = Cells(1, kColTasaCuotaSegundoIEPS1).ColumnWidth
    
        Range("FProceso").Value = "Preparando empresa..."
        'hDbc = dbLogIn("Usuario MS Excel", Trim(Worksheets("Parámetros").Range("UbicacionEmpresa").Value))
        Call comFacturaMain.dbLogIn("Usuario MS Excel", Trim(Worksheets("Parámetros").Range("UbicacionEmpresa").Value), hDbc)
        'lError = fAbreEmpresa(Trim(Worksheets("Parámetros").Range("UbicacionEmpresa").Value))
        Call comFacturaMain.AbreEmpresa(Trim(Worksheets("Parámetros").Range("UbicacionEmpresa").Value), lError)
        If lError = 0 Then
            'lError = fConfiguraEmpresaFacturaGlobal(lCodigoConcepto, lCodigoServicio)
            Call comFacturaMain.ConfiguraEmpresaFacturaGlobal(lCodigoConcepto, lCodigoServicio, lError)
        End If
        If lError <> 0 Then
           MsgBox MensajeError(lError), vbOKOnly
        End If
        'Call fCierraEmpresa
        Call comFacturaMain.CierraEmpresa
        If Range("FormaDePago") = "" Then
            Range("FormaDePago").Value = "01=Efectivo"
        End If
        Range("CodigoConcepto").Value = Trim(fEliminaEspaciosNulos2(lCodigoConcepto))
        Range("CodigoServicio").Value = Trim(fEliminaEspaciosNulos2(lCodigoServicio))
        Range("FProceso").Value = ""
        
        Range("FFecha").Calculate
    
        'dbGetTopNoLock hDbc, PARAMETROS, IxPARAMETROS
        'dbFieldNameLong hDbc, PARAMETROS, "cPrecios01", lIVAIncluido
        Call comFacturaMain.dbGetTopNoLock(hDbc, PARAMETROS, IxPARAMETROS, lResultado)
        Call comFacturaMain.dbFieldNameLong(hDbc, PARAMETROS, "cPrecios01", lIVAIncluido, lResultado)
        Cells(kRenIVAIncluido, kColIVAIncluido).Value = lIVAIncluido
        If lIVAIncluido = 0 Then
            Cells(4, 5).Value = "Empresa configurada con Precios de venta sin IVA incluido"
            Cells(7, 5).Value = "El precio unitario de cada ticket correspondera a la suma de las bases más el IVA"
        Else
            Cells(4, 5).Value = "Empresa configurada con Precios de venta con IVA incluido"
            Cells(7, 5).Value = "El precio unitario de cada ticket correspondera a la suma de las bases"
        End If
        
    End If
    Exit Sub
    
Error_Worksheet_Activate:
    Range("FProceso").Value = ""
    MsgBox Err.Description, vbExclamation + vbOKOnly
End Sub


Private Sub Worksheet_Change(ByVal Target As Range)
Dim lTotal As Double
Dim lTotalCascada As Double 'Este es el total cuando se realizan cálculos de impuestos en cascada y se tienen múltiples tasas o cuotas en una sola nota de venta
Dim lIva As Double
Dim lPrecio As Double
Dim lImpRetLocal As Double
Dim lIEPS As Double
Dim lSegundoIEPS As Double
Dim lIEPSTasaIVAGeneral As Double
Dim lSegundoIEPSTasaIVAGeneral As Double
Dim lBaseIEPS As Double
Dim lCampoCalculoIEPS As String * 10
Dim lFormulaIEPS As Long
Dim lFormulaIVA As Long
Dim lEsCalculoCascada As Boolean
Dim lRenglon As Long
Dim lTasaCuotaIEPSActual As Integer
Dim lImpuesto As Long
Dim lCampoImpRetLocal As String
Dim lTasaImpRetCapturada As Boolean
Dim lIvaGeneralNoCapturado As Boolean
Dim lResultado As Long
                
    If Not Intersect(Target, Range("D22:K2022")) Is Nothing Or Not Intersect(Target, Range("M22:M2022")) Is Nothing Or Not Intersect(Target, Range("O22:O2022")) Is Nothing Or Not Intersect(Target, Range("Q22:Q2022")) Is Nothing Or Not Intersect(Target, Range("S22:S2022")) Is Nothing Or Not Intersect(Target, Range("U22:U2022")) Is Nothing Then
        For lRenglon = Target.Row To (Target.Row + Target.Rows.Count - 1)
            lTotal = 0#
            lIva = 0#
            lImpRetLocal = 0#
            lPrecio = 0#
            lTasaImpRetCapturada = False
            lIvaGeneralNoCapturado = False
            
            If CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0 Then
                If Cells(kRenIVAIncluido, kColIVAIncluido).Value = 0 Then
                    lPrecio = Cells(lRenglon, kColIvaGeneral).Value
                    lIva = (Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaIVA
                    lTotal = (Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) + ((Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaIVA)
                Else
                    lPrecio = (Cells(lRenglon, kColIvaGeneral).Value / (1 + gTasaIVA))
                    lIva = ((Cells(lRenglon, kColIvaGeneral).Value / (1 + gTasaIVA)) - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaIVA
                    lTotal = (Cells(lRenglon, kColIvaGeneral).Value / (1 + gTasaIVA)) - Cells(lRenglon, kColIvaGeneral + 1).Value + lIva
                End If
            End If
            If CDbl(Cells(lRenglon, kColIvaTasa0).Value) <> 0 Then
                lTotal = lTotal + (Cells(lRenglon, kColIvaTasa0).Value - Cells(lRenglon, kColIvaTasa0 + 1).Value)
                lPrecio = lPrecio + Cells(lRenglon, kColIvaTasa0).Value
            End If
            If CDbl(Cells(lRenglon, kColIvaExento).Value) <> 0 Then
                lTotal = lTotal + (Cells(lRenglon, kColIvaExento).Value - Cells(lRenglon, kColIvaExento + 1).Value)
                lPrecio = lPrecio + Cells(lRenglon, kColIvaExento).Value
            End If
            
            If gNumeroImpRetLocal <> 0 Then
                lCampoImpRetLocal = "cCfdImpu0" & gNumeroImpRetLocal
                'dbGetTopNoLock hDbc, PARAMETROS, IxPARAMETROS
                'dbFieldNameLong hDbc, PARAMETROS, lCampoImpRetLocal, lImpuesto
                Call comFacturaMain.dbGetTopNoLock(hDbc, PARAMETROS, IxPARAMETROS, lResultado)
                Call comFacturaMain.dbFieldNameLong(hDbc, PARAMETROS, lCampoImpRetLocal, lImpuesto, lResultado)
                If Cells(lRenglon, kColImpuestoLocal).Value = "" Then
                    If Not Worksheets("Parámetros").Range("TasaImpRetLocal").Value = "" Then
                        If CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0 Then
                            Cells(lRenglon, kColImpuestoLocal).Value = Worksheets("Parámetros").Range("TasaImpRetLocal").Value
                            lImpRetLocal = (Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaImpRetLocal
                        End If
                        If CDbl(Cells(lRenglon, kColIvaTasa0).Value) <> 0 Then
                            Cells(lRenglon, kColImpuestoLocal).Value = Worksheets("Parámetros").Range("TasaImpRetLocal").Value
                            lImpRetLocal = lImpRetLocal + ((Cells(lRenglon, kColIvaTasa0).Value - Cells(lRenglon, kColIvaTasa0 + 1).Value) * gTasaImpRetLocal)
                        End If
                        If lImpuesto = 5 Then
                            lTotal = lTotal - lImpRetLocal
                        Else
                            lTotal = lTotal + lImpRetLocal
                        End If
                    End If
                Else
                'kOpcionesIVA = "1=General,2=General/Exento,3=General/Tasa0%,4=General/Tasa0%/Exento"
                    If gIncluirTasasIVA = 1 Or gIncluirTasasIVA = 2 Then
                        If CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0 Then
                            lImpRetLocal = (Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaImpRetLocal
                            If lImpuesto = 5 Then
                                lTotal = lTotal - lImpRetLocal
                            Else
                                lTotal = lTotal + lImpRetLocal
                            End If
                        Else
                            Cells(lRenglon, kColImpuestoLocal).Value = ""
                        End If
                    Else
                        If gIncluirTasasIVA = 3 Or gIncluirTasasIVA = 4 Then
                            If CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0 Then
                                lImpRetLocal = (Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaImpRetLocal
                            Else
                                lIvaGeneralNoCapturado = True
                            End If
                            
                            If CDbl(Cells(lRenglon, kColIvaTasa0).Value) <> 0 Then
                                lImpRetLocal = lImpRetLocal + ((Cells(lRenglon, kColIvaTasa0).Value - Cells(lRenglon, kColIvaTasa0 + 1).Value) * gTasaImpRetLocal)
                            Else
                                If lIvaGeneralNoCapturado Then
                                    Cells(lRenglon, kColImpuestoLocal).Value = ""
                                End If
                            End If
                            If lImpuesto = 5 Then
                                lTotal = lTotal - lImpRetLocal
                            Else
                                lTotal = lTotal + lImpRetLocal
                            End If
                        End If
                    End If
                End If
            End If
            
            If gNumeroImpuestoIEPS <> 0 Then
                lTasaCuotaIEPSActual = 0
                lIEPS = 0
                lIEPSTasaIVAGeneral = 0
                lEsCalculoCascada = False
                lTotalCascada = lTotal
                If CDbl(Cells(lRenglon, kColIvaExento).Value) <> 0 Then
                    lTotalCascada = lTotalCascada - (Cells(lRenglon, kColIvaExento).Value - Cells(lRenglon, kColIvaExento + 1).Value)
                End If
                While lTasaCuotaIEPSActual < CInt(Worksheets("Parámetros").Range("CantidadTasasCuotas").Value)
                    If CDbl(Cells(lRenglon, kColTasaCuotaIEPS1 + (lTasaCuotaIEPSActual * 2)).Value) <> 0 Then
                        If (Not CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0) And (Not CDbl(Cells(lRenglon, kColIvaTasa0).Value) <> 0) And CDbl(Cells(lRenglon, kColIvaExento).Value) <> 0 Then
                            Cells(lRenglon, kColTasaCuotaIEPS1 + (lTasaCuotaIEPSActual * 2)).Value = ""
                            Cells(lRenglon, kColBaseIEPS1 + (lTasaCuotaIEPSActual * 2)).Value = ""
                        Else
                            If Worksheets("Parámetros").Range("TasaOCuotaIEPS").Value = "Tasa" Then
                                gTasaIEPS = CDbl(Cells(lRenglon, kColTasaCuotaIEPS1 + (lTasaCuotaIEPSActual * 2)).Value) / 100
                                'If dbGetNoLock(hDbc, CONCEPTOS, IxCCODIGOC01, Range("CodigoConcepto").Value) = 0 Then
                                Call comFacturaMain.dbGetNoLock(hDbc, CONCEPTOS, IxCCODIGOC01, Range("CodigoConcepto").Value, lResultado)
                                If lResultado = 0 Then
                                    If gNumeroImpuestoIEPS = 1 Then
                                       lCampoCalculoIEPS = "cIdFormu05"
                                    Else
                                        If gNumeroImpuestoIEPS = 2 Then
                                            lCampoCalculoIEPS = "cIdFormu07"
                                        End If
                                    End If
                                    'dbFieldNameLong hDbc, CONCEPTOS, lCampoCalculoIEPS, lFormulaIEPS
                                    Call comFacturaMain.dbFieldNameLong(hDbc, CONCEPTOS, lCampoCalculoIEPS, lFormulaIEPS, lResultado)
                                    If gNumeroImpuestoIEPS = 1 Or lFormulaIEPS = 61 Or lFormulaIEPS = 51 Or lFormulaIEPS = 64 Then
                                        If CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0 And CDbl(Cells(lRenglon, kColIvaTasa0).Value) <> 0 Then
                                            lIEPSTasaIVAGeneral = lIEPSTasaIVAGeneral + ((Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaIEPS)
                                            lIEPS = lIEPS + ((Cells(lRenglon, kColIvaTasa0).Value - Cells(lRenglon, kColIvaTasa0 + 1).Value) * gTasaIEPS)
                                            lBaseIEPS = Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value + Cells(lRenglon, kColIvaTasa0).Value - Cells(lRenglon, kColIvaTasa0 + 1).Value
                                        Else
                                            If CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0 Then
                                                lIEPSTasaIVAGeneral = lIEPSTasaIVAGeneral + ((Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaIEPS)
                                                'lIEPS = lIEPS + lIEPSTasaIVAGeneral
                                                lBaseIEPS = Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value
                                            Else
                                                lIEPS = lIEPS + ((Cells(lRenglon, kColIvaTasa0).Value - Cells(lRenglon, kColIvaTasa0 + 1).Value) * gTasaIEPS)
                                                lBaseIEPS = Cells(lRenglon, kColIvaTasa0).Value - Cells(lRenglon, kColIvaTasa0 + 1).Value
                                            End If
                                        End If
                                    Else
                                        lEsCalculoCascada = True
                                        lIEPS = lIEPS + (lTotalCascada * gTasaIEPS)
                                        lBaseIEPS = lTotalCascada
                                        lIEPSTasaIVAGeneral = lIEPS
                                    End If
                                End If
                            Else
                                lBaseIEPS = 1
                                If CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0 Then
                                    lIEPSTasaIVAGeneral = lIEPSTasaIVAGeneral + CDbl(Cells(lRenglon, kColTasaCuotaIEPS1 + (lTasaCuotaIEPSActual * 2)).Value)
                                Else
                                    lIEPS = lIEPS + CDbl(Cells(lRenglon, kColTasaCuotaIEPS1 + (lTasaCuotaIEPSActual * 2)).Value)
                                End If
                            End If
                            'Cells(lRenglon, kColTotal + 3).Value = lIEPS
                            If Not Cells(lRenglon, kColBaseIEPS1 + (lTasaCuotaIEPSActual * 2)).Value <> 0 Then
                                Cells(lRenglon, kColBaseIEPS1 + (lTasaCuotaIEPSActual * 2)).Value = lBaseIEPS
                            End If
                        End If
                    End If
                    lTasaCuotaIEPSActual = lTasaCuotaIEPSActual + 1
                Wend
                
                'If dbGetNoLock(hDbc, CONCEPTOS, IxCCODIGOC01, Range("CodigoConcepto").Value) = 0 Then
                Call comFacturaMain.dbGetNoLock(hDbc, CONCEPTOS, IxCCODIGOC01, Range("CodigoConcepto").Value, lResultado)
                If lResultado = 0 Then
                    'dbFieldNameLong hDbc, CONCEPTOS, "cIdFormu07", lFormulaIVA
                    If gNumeroImpuestoIVA = 2 Then
                        Call comFacturaMain.dbFieldNameLong(hDbc, CONCEPTOS, "cIdFormu07", lFormulaIVA, lResultado)
                    Else
                        Call comFacturaMain.dbFieldNameLong(hDbc, CONCEPTOS, "cIdFormu09", lFormulaIVA, lResultado)
                    End If
                    If (gNumeroImpuestoIVA = 2 And lFormulaIVA = 63) Or (gNumeroImpuestoIVA = 3 And lFormulaIVA = 73) Then
                        lTotal = lTotal - ((Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaIVA)
                        lIva = Round((Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value + lIEPSTasaIVAGeneral) * gTasaIVA, 2)
                        lTotal = lTotal + lIva
                    End If
                End If
                'Carlos inicio
                If lEsCalculoCascada = False Then
                    lIEPS = lIEPS + lIEPSTasaIVAGeneral
                End If
                'Carlos fin
                'lIEPS = lIEPS + lIEPSTasaIVAGeneral
                Cells(lRenglon, kColTotal + 3).Value = lIEPS
                lTotal = lTotal + lIEPS
            End If
            
            If gNumeroSegundoImpuestoIEPS <> 0 Then
                lTasaCuotaIEPSActual = 0
                lSegundoIEPS = 0
                lSegundoIEPSTasaIVAGeneral = 0
                lEsCalculoCascada = False
                If gNumeroImpuestoIVA = 3 Then
                    lTotalCascada = lTotal - lIva
                Else
                    lTotalCascada = lTotal
                End If
                If CDbl(Cells(lRenglon, kColIvaExento).Value) <> 0 Then
                    lTotalCascada = lTotalCascada - (Cells(lRenglon, kColIvaExento).Value - Cells(lRenglon, kColIvaExento + 1).Value)
                End If
                While lTasaCuotaIEPSActual < CInt(Worksheets("Parámetros").Range("CantidadTasasCuotasSegundoIEPS").Value)
                    If CDbl(Cells(lRenglon, kColTasaCuotaSegundoIEPS1 + (lTasaCuotaIEPSActual * 2)).Value) <> 0 Then
                        If (Not CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0) And (Not CDbl(Cells(lRenglon, kColIvaTasa0).Value) <> 0) And CDbl(Cells(lRenglon, kColIvaExento).Value) <> 0 Then
                            Cells(lRenglon, kColTasaCuotaSegundoIEPS1 + (lTasaCuotaIEPSActual * 2)).Value = ""
                            Cells(lRenglon, kColBaseSegundoIEPS1 + (lTasaCuotaIEPSActual * 2)).Value = ""
                        Else
                            If Worksheets("Parámetros").Range("TasaOCuotaSegundoIEPS").Value = "Tasa" Then
                                gTasaSegundoIEPS = CDbl(Cells(lRenglon, kColTasaCuotaSegundoIEPS1 + (lTasaCuotaIEPSActual * 2)).Value) / 100
                                'If dbGetNoLock(hDbc, CONCEPTOS, IxCCODIGOC01, Range("CodigoConcepto").Value) = 0 Then
                                Call comFacturaMain.dbGetNoLock(hDbc, CONCEPTOS, IxCCODIGOC01, Range("CodigoConcepto").Value, lResultado)
                                If lResultado = 0 Then
                                    If gNumeroSegundoImpuestoIEPS = 2 Then
                                       lCampoCalculoIEPS = "cIdFormu07"
                                    Else
                                        If gNumeroSegundoImpuestoIEPS = 3 Then
                                            lCampoCalculoIEPS = "cIdFormu09"
                                        End If
                                    End If
                                    'dbFieldNameLong hDbc, CONCEPTOS, lCampoCalculoIEPS, lFormulaIEPS
                                    Call comFacturaMain.dbFieldNameLong(hDbc, CONCEPTOS, lCampoCalculoIEPS, lFormulaIEPS, lResultado)
                                    If lFormulaIEPS = 61 Or lFormulaIEPS = 64 Or lFormulaIEPS = 71 Then
                                        If CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0 And CDbl(Cells(lRenglon, kColIvaTasa0).Value) <> 0 Then
                                            lSegundoIEPSTasaIVAGeneral = lSegundoIEPSTasaIVAGeneral + ((Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaSegundoIEPS)
                                            lSegundoIEPS = lSegundoIEPS + ((Cells(lRenglon, kColIvaTasa0).Value - Cells(lRenglon, kColIvaTasa0 + 1).Value) * gTasaSegundoIEPS)
                                            lBaseIEPS = Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value + Cells(lRenglon, kColIvaTasa0).Value - Cells(lRenglon, kColIvaTasa0 + 1).Value
                                        Else
                                            If CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0 Then
                                                lSegundoIEPSTasaIVAGeneral = lSegundoIEPSTasaIVAGeneral + ((Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaSegundoIEPS)
                                                'lIEPS = lIEPS + lIEPSTasaIVAGeneral
                                                lBaseIEPS = Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value
                                            Else
                                                lSegundoIEPS = lSegundoIEPS + ((Cells(lRenglon, kColIvaTasa0).Value - Cells(lRenglon, kColIvaTasa0 + 1).Value) * gTasaSegundoIEPS)
                                                lBaseIEPS = Cells(lRenglon, kColIvaTasa0).Value - Cells(lRenglon, kColIvaTasa0 + 1).Value
                                            End If
                                        End If
                                    Else
                                        lEsCalculoCascada = True
                                        lSegundoIEPS = lSegundoIEPS + (lTotalCascada * gTasaSegundoIEPS)
                                        lBaseIEPS = lTotalCascada
                                        lSegundoIEPSTasaIVAGeneral = lSegundoIEPS
                                    End If
                                End If
                            Else
                                lBaseIEPS = 1
                                If CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0 Then
                                    lSegundoIEPSTasaIVAGeneral = lSegundoIEPSTasaIVAGeneral + CDbl(Cells(lRenglon, kColTasaCuotaSegundoIEPS1 + (lTasaCuotaIEPSActual * 2)).Value)
                                Else
                                    lSegundoIEPS = lSegundoIEPS + CDbl(Cells(lRenglon, kColTasaCuotaSegundoIEPS1 + (lTasaCuotaIEPSActual * 2)).Value)
                                End If
                            End If
                            'Cells(lRenglon, kColTotal + 3).Value = lIEPS
                            If Not Cells(lRenglon, kColBaseSegundoIEPS1 + (lTasaCuotaIEPSActual * 2)).Value <> 0 Then
                                Cells(lRenglon, kColBaseSegundoIEPS1 + (lTasaCuotaIEPSActual * 2)).Value = lBaseIEPS
                            End If
                        End If
                    End If
                    lTasaCuotaIEPSActual = lTasaCuotaIEPSActual + 1
                Wend
                
                'If dbGetNoLock(hDbc, CONCEPTOS, IxCCODIGOC01, Range("CodigoConcepto").Value) = 0 Then
                Call comFacturaMain.dbGetNoLock(hDbc, CONCEPTOS, IxCCODIGOC01, Range("CodigoConcepto").Value, lResultado)
                If lResultado = 0 Then
                    'dbFieldNameLong hDbc, CONCEPTOS, "cIdFormu07", lFormulaIVA
                    Call comFacturaMain.dbFieldNameLong(hDbc, CONCEPTOS, "cIdFormu09", lFormulaIVA, lResultado)
                    If gNumeroImpuestoIVA = 3 And lFormulaIVA = 73 And CDbl(Cells(lRenglon, kColIvaGeneral).Value) <> 0 Then
                        'lTotal = lTotal - ((Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value) * gTasaIVA)
                        lTotal = lTotal - lIva
                        lIva = Round((Cells(lRenglon, kColIvaGeneral).Value - Cells(lRenglon, kColIvaGeneral + 1).Value + lIEPSTasaIVAGeneral + lSegundoIEPSTasaIVAGeneral) * gTasaIVA, 2)
                        lTotal = lTotal + lIva
                    End If
                End If
                If lEsCalculoCascada = False Then
                    lSegundoIEPS = lSegundoIEPS + lSegundoIEPSTasaIVAGeneral
                End If
                'lSegundoIEPS = lSegundoIEPS + lSegundoIEPSTasaIVAGeneral
                Cells(lRenglon, kColTotal + 4).Value = lSegundoIEPS
                lTotal = lTotal + lSegundoIEPS
            End If
            
            Cells(lRenglon, kColTotal + 5).Value = ""
            If lTotal <= 0 Then
                Cells(lRenglon, kColTotal + 5).Value = "El importe del movimiento no puede ser cero o negativo (" & lTotal & ")"
                lTotal = 0
                lIva = 0
            End If
            Cells(lRenglon, kColTotal).Value = lTotal
            Cells(lRenglon, kColTotal + 1).Value = lIva
            Cells(lRenglon, kColTotal + 2).Value = lImpRetLocal
            Cells(lRenglon, kColTotal + 5).Value = lPrecio
            Cells(lRenglon, kColTotal).NumberFormat = "0.00"
            Cells(lRenglon, kColTotal + 1).NumberFormat = "0.00"
            Cells(lRenglon, kColTotal + 2).NumberFormat = "0.00"
            Cells(lRenglon, kColTotal + 3).NumberFormat = "0.00"
            Cells(lRenglon, kColTotal + 4).NumberFormat = "0.00"
            Cells(lRenglon, kColTotal + 5).NumberFormat = "0.00"
        Next
    End If
End Sub

Public Sub DesplegarCatalogoF3()
    If ActiveSheet.Name <> "Factura" Then
        GoTo fin
    End If
    ' se establece con cual función se trabaja
    On Error GoTo hdlerr
    
    If Not Intersect(ActiveCell, ActiveSheet.Range("CodigoConcepto")) Is Nothing Then
        Call FbtnConcepto_Click
        GoTo fin
    End If
    If Not Intersect(ActiveCell, ActiveSheet.Range("CodigoCliente")) Is Nothing Then
        Call FbtnCliente_Click
        GoTo fin
    End If
    If Not Intersect(ActiveCell, ActiveSheet.Range("CodigoServicio")) Is Nothing Then
        Call FbtnServicio_Click
        GoTo fin
    End If
    If Not Intersect(ActiveCell, ActiveSheet.Range("FormaDePago")) Is Nothing Then
        Call FbtnFormaPago_Click
        GoTo fin
    End If
    GoTo fin
hdlerr:
    MsgBox Err.Description
fin:
End Sub
